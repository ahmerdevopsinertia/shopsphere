import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ConfigService } from '@nestjs/config';
import helmet from 'helmet';


async function bootstrap() {

	const app =
		await NestFactory.create(
			AppModule,
		);


	// Security Headers
	app.use(
		helmet(),
	);


	const configService =
		app.get(ConfigService);


	// CORS
	app.enableCors({
		origin: configService.get<string>(
			'app.corsOrigin',
		),
		credentials: true,
	});


	const port =
		configService.get<number>(
			'app.port',
		);


	await app.listen(
		port ?? 3000,
	);
}


bootstrap();

Step 1 — Verify Global ValidationPipe

Open:

src/main.ts

Confirm you have:

app.useGlobalPipes(
	new ValidationPipe({
		whitelist: true,
		forbidNonWhitelisted: true,
		transform: true,
	}),
);
Why?
whitelist

Removes unknown fields.

Example request:

{
  "email": "test@test.com",
  "password": "Password1",
  "isAdmin": true
}

Without whitelist:

{
  "email": "test@test.com",
  "password": "Password1",
  "isAdmin": true
}

With whitelist:

{
  "email": "test@test.com",
  "password": "Password1"
}
forbidNonWhitelisted

Instead of silently removing:

Request:

{
 "email":"test@test.com",
 "hackField":"123"
}

Response:

{
 "success":false,
 "statusCode":400,
 "message":"property hackField should not exist"
}

This is better for security.

For ShopSphere, I recommend not running seed on every production container restart because it can create duplicate data. Instead:

migrate deploy → always runs ✅
seed → run only when explicitly required ✅
Step 1 — Create Prisma Seed File

Create:

prisma/seed.ts

Example:

import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

async function main() {

  console.log('🌱 Starting database seed...');


  // Create category
  const category = await prisma.category.upsert({
    where: {
      name: 'Mobile Accessories',
    },
    update: {},
    create: {
      name: 'Mobile Accessories',
    },
  });


  // Create product
  await prisma.product.upsert({
    where: {
      name: 'iPhone 17 Case',
    },
    update: {},
    create: {
      name: 'iPhone 17 Case',
      price: 39,
      categoryId: category.id,
    },
  });


  console.log('🌱 Database seed completed');
}


main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });

Step 4: Add global exception logging filter

Create:

src/common/filters/all-exceptions.filter.ts

Add:

import {
	ArgumentsHost,
	Catch,
	ExceptionFilter,
	HttpException,
	Logger,
} from '@nestjs/common';


@Catch()
export class AllExceptionsFilter 
implements ExceptionFilter {


	private readonly logger = new Logger(
		'GlobalException'
	);


	catch(exception: unknown, host: ArgumentsHost) {


		const ctx = host.switchToHttp();

		const response = ctx.getResponse();

		const request = ctx.getRequest();


		const status =
			exception instanceof HttpException
				? exception.getStatus()
				: 500;



		const message =
			exception instanceof HttpException
				? exception.message
				: 'Internal server error';



		this.logger.error({

			method: request.method,

			url: request.url,

			statusCode: status,

			message,

			stack:
				exception instanceof Error
					? exception.stack
					: undefined,

		});



		response.status(status).json({

			success:false,

			statusCode:status,

			message,

			timestamp:new Date().toISOString(),

			path:request.url,

		});


	}

}
Step 5: Register filter globally in main.ts

Add:

import { AllExceptionsFilter } from './common/filters/all-exceptions.filter';


async function bootstrap() {

 const app = await NestFactory.create(
   AppModule,
   {
     bufferLogs:true
   }
 );


 app.useLogger(app.get(Logger));


 app.useGlobalFilters(
    new AllExceptionsFilter()
 );


 await app.listen(3000);

}

Good catch. 👍 Before freezing Sprint 8 and moving to CI/CD, I agree we should not leave authentication incomplete.

A production-ready authentication flow should include:

✅ Register
✅ Login
✅ Access Token generation
❌ Refresh Token API (missing)
❌ Logout / revoke refresh token (future enhancement)

Since we already made architectural decisions earlier:

ADR-005: Access Token + Refresh Token Strategy
ADR-006: Hash Refresh Tokens Before Storing

we should implement the refresh token flow now and then create the Sprint 8 tag after that.

Current Auth Flow

Currently you probably have:

User
 |
 | email + password
 ↓
POST /auth/login
 |
 ↓
Validate credentials
 |
 ↓
Generate JWT Access Token
 |
 ↓
Return token

The problem:

Access tokens are short-lived. Example:

ACCESS_TOKEN_EXPIRES_IN=15m

After 15 minutes the user is forced to login again.

Target Production Flow

We should implement:

POST /auth/login

Response:

{
  "accessToken": "...",
  "refreshToken": "..."
}

Then:

POST /auth/refresh

Request:

{
  "refreshToken": "xxxx"
}


Response:

{
  "accessToken": "new token",
  "refreshToken": "new refresh token"
}
Database Change Required

Your current User model:

model User {
  id           String   @id @default(uuid())
  email        String   @unique
  passwordHash String
  role         Role     @default(CUSTOMER)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  profile UserProfile?
  orders Order[]
  cart Cart?
}

We should add a refresh token table instead of storing it directly.

Why?

Because later we can support:

multiple devices
revoke individual sessions
logout from all devices

Add:

model RefreshToken {

  id String @id @default(uuid())

  tokenHash String

  userId String

  user User @relation(
    fields:[userId],
    references:[id],
    onDelete:Cascade
  )

  expiresAt DateTime

  revoked Boolean @default(false)

  createdAt DateTime @default(now())

  updatedAt DateTime @updatedAt

  @@index([userId])
}

Then migration:

npx prisma migrate dev --name add_refresh_token
Auth Service Changes

Login becomes:

async login(dto: LoginDto){

 const user =
   await this.usersRepository.findByEmail(dto.email);


 if(!user){
   throw new UnauthorizedException();
 }


 const passwordMatch =
   await bcrypt.compare(
      dto.password,
      user.passwordHash
   );


 if(!passwordMatch){
    throw new UnauthorizedException();
 }


 const tokens =
   await this.generateTokens(user);


 await this.saveRefreshToken(
    user.id,
    tokens.refreshToken
 );


 return tokens;
}
Token Generation

Create:

private async generateTokens(user){

 const payload={
   sub:user.id,
   email:user.email,
   role:user.role
 };


 const accessToken =
 await this.jwtService.signAsync(
    payload,
    {
      expiresIn:'15m'
    }
 );


 const refreshToken =
 await this.jwtService.signAsync(
    payload,
    {
      expiresIn:'7d'
    }
 );


 return {
    accessToken,
    refreshToken
 };
}
Important Part: Hash Refresh Token

Following ADR-006:

Never store:

eyJhbGciOiJIUz...

directly.

Store:

8f7a92a3d8c....

Example:

const hash =
 await bcrypt.hash(
    refreshToken,
    10
 );


await prisma.refreshToken.create({
 data:{
   userId:user.id,
   tokenHash:hash,
   expiresAt:
    new Date(Date.now()+7*24*60*60*1000)
 }
});
